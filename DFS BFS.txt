#include <iostream>
#include <vector>
#include <queue>
#include <stack>
#include <chrono>
#include <omp.h>

using namespace std;

class Graph {

    int vertices;
    int edges;

    vector<vector<int>> graph;
    vector<bool> visited;

public:

    // ===================== CONSTRUCTOR =====================

    Graph() {

        cout << "Enter number of vertices: ";
        cin >> vertices;

        cout << "Enter number of edges: ";
        cin >> edges;

        graph.assign(vertices, vector<int>());

        visited.assign(vertices, false);

        for(int i = 0; i < edges; i++) {

            int a, b;

            cout << "Enter edge " << i + 1 << ": ";
            cin >> a >> b;

            addEdge(a, b);
        }
    }

    // ===================== ADD EDGE =====================

    void addEdge(int a, int b) {

        graph[a].push_back(b);
        graph[b].push_back(a);
    }

    // ===================== PRINT GRAPH =====================

    void printGraph() {

        cout << "\nAdjacency List:\n";

        for(int i = 0; i < vertices; i++) {

            cout << i << " -> ";

            for(auto j : graph[i]) {
                cout << j << " ";
            }

            cout << endl;
        }
    }

    // ===================== RESET VISITED =====================

    void initialize_visited() {

        visited.assign(vertices, false);
    }

    // ===================== SEQUENTIAL DFS =====================

    void dfs(int start) {

        stack<int> s;

        s.push(start);

        visited[start] = true;

        while(!s.empty()) {

            int current = s.top();

            s.pop();

            cout << current << " ";

            for(auto neighbor : graph[current]) {

                if(!visited[neighbor]) {

                    visited[neighbor] = true;

                    s.push(neighbor);
                }
            }
        }
    }

    // ===================== PARALLEL DFS =====================

    void parallel_dfs(int start) {

        stack<int> s;

        s.push(start);

        visited[start] = true;

        while(!s.empty()) {

            int current;

            #pragma omp critical
            {
                current = s.top();
                s.pop();
            }

            cout << current << " ";

            #pragma omp parallel for
            for(int i = 0; i < graph[current].size(); i++) {

                int neighbor = graph[current][i];

                #pragma omp critical
                {
                    if(!visited[neighbor]) {

                        visited[neighbor] = true;

                        s.push(neighbor);
                    }
                }
            }
        }
    }

    // ===================== SEQUENTIAL BFS =====================

    void bfs(int start) {

        queue<int> q;

        q.push(start);

        visited[start] = true;

        while(!q.empty()) {

            int current = q.front();

            q.pop();

            cout << current << " ";

            for(auto neighbor : graph[current]) {

                if(!visited[neighbor]) {

                    visited[neighbor] = true;

                    q.push(neighbor);
                }
            }
        }
    }

    // ===================== PARALLEL BFS =====================

    void parallel_bfs(int start) {

        queue<int> q;

        q.push(start);

        visited[start] = true;

        while(!q.empty()) {

            int current;

            #pragma omp critical
            {
                current = q.front();
                q.pop();
            }

            cout << current << " ";

            #pragma omp parallel for
            for(int i = 0; i < graph[current].size(); i++) {

                int neighbor = graph[current][i];

                #pragma omp critical
                {
                    if(!visited[neighbor]) {

                        visited[neighbor] = true;

                        q.push(neighbor);
                    }
                }
            }
        }
    }
};

// ===================== MAIN FUNCTION =====================

int main() {

    Graph g;

    g.printGraph();

    // ===================== DFS =====================

    g.initialize_visited();

    cout << "\nSequential DFS:\n";

    auto start = chrono::high_resolution_clock::now();

    g.dfs(0);

    auto end = chrono::high_resolution_clock::now();

    cout << "\nTime Taken: "
         << chrono::duration_cast<chrono::microseconds>(end - start).count()
         << " microseconds\n";

    // ===================== PARALLEL DFS =====================

    g.initialize_visited();

    cout << "\nParallel DFS:\n";

    start = chrono::high_resolution_clock::now();

    g.parallel_dfs(0);

    end = chrono::high_resolution_clock::now();

    cout << "\nTime Taken: "
         << chrono::duration_cast<chrono::microseconds>(end - start).count()
         << " microseconds\n";

    // ===================== BFS =====================

    g.initialize_visited();

    cout << "\nSequential BFS:\n";

    start = chrono::high_resolution_clock::now();

    g.bfs(0);

    end = chrono::high_resolution_clock::now();

    cout << "\nTime Taken: "
         << chrono::duration_cast<chrono::microseconds>(end - start).count()
         << " microseconds\n";

    // ===================== PARALLEL BFS =====================

    g.initialize_visited();

    cout << "\nParallel BFS:\n";

    start = chrono::high_resolution_clock::now();

    g.parallel_bfs(0);

    end = chrono::high_resolution_clock::now();

    cout << "\nTime Taken: "
         << chrono::duration_cast<chrono::microseconds>(end - start).count()
         << " microseconds\n";

    return 0;
}
