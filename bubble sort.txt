// ===================== SEQUENTIAL BUBBLE SORT =====================

void sequential_bubble_sort(int arr[], int size){

    vector<int> array(size);

    for(int i = 0; i < size; i++){
        array[i] = arr[i];
    }

    double start = omp_get_wtime();

    for(int i = 0; i < size - 1; i++){

        for(int j = 0; j < size - i - 1; j++){

            if(array[j] > array[j + 1]){
                swap(array[j], array[j + 1]);
            }
        }
    }

    double end = omp_get_wtime();

    cout << "Sequential Bubble Sort:\n";
    cout << "Time Required: " << end - start << " seconds\n\n";
}

// ===================== PARALLEL BUBBLE SORT =====================

void parallel_bubble_sort(int arr[], int size){

    vector<int> array(size);

    for(int i = 0; i < size; i++){
        array[i] = arr[i];
    }

    double start = omp_get_wtime();

    for(int k = 0; k < size; k++){

        // EVEN PHASE
        if(k % 2 == 0){

            #pragma omp parallel for
            for(int i = 1; i < size - 1; i += 2){

                if(array[i] > array[i + 1]){
                    swap(array[i], array[i + 1]);
                }
            }
        }

        // ODD PHASE
        else{

            #pragma omp parallel for
            for(int i = 0; i < size - 1; i += 2){

                if(array[i] > array[i + 1]){
                    swap(array[i], array[i + 1]);
                }
            }
        }
    }

    double end = omp_get_wtime();

    cout << "Parallel Bubble Sort:\n";
    cout << "Time Required: " << end - start << " seconds\n\n";
}

// ===================== MERGE FUNCTION =====================

void merge(int arr[], int low, int mid, int high){

    vector<int> temp(high - low + 1);

    int i = low;
    int j = mid + 1;
    int k = 0;

    while(i <= mid && j <= high){

        if(arr[i] <= arr[j]){
            temp[k++] = arr[i++];
        }
        else{
            temp[k++] = arr[j++];
        }
    }

    while(i <= mid){
        temp[k++] = arr[i++];
    }

    while(j <= high){
        temp[k++] = arr[j++];
    }

    for(i = low, k = 0; i <= high; i++, k++){
        arr[i] = temp[k];
    }
}

// ===================== SEQUENTIAL MERGE SORT =====================

void mergesort(int arr[], int low, int high){

    if(low < high){

        int mid = (low + high) / 2;

        mergesort(arr, low, mid);

        mergesort(arr, mid + 1, high);

        merge(arr, low, mid, high);
    }
}

void perform_merge_sort(int arr[], int size){

    vector<int> array(size);

    for(int i = 0; i < size; i++){
        array[i] = arr[i];
    }

    double start = omp_get_wtime();

    mergesort(array.data(), 0, size - 1);

    double end = omp_get_wtime();

    cout << "Sequential Merge Sort:\n";
    cout << "Time Required: " << end - start << " seconds\n\n";
}

// ===================== PARALLEL MERGE SORT =====================

void p_mergesort(int arr[], int low, int high){

    if(low < high){

        int mid = (low + high) / 2;

        #pragma omp parallel sections
        {

            #pragma omp section
            {
                p_mergesort(arr, low, mid);
            }

            #pragma omp section
            {
                p_mergesort(arr, mid + 1, high);
            }
        }

        merge(arr, low, mid, high);
    }
}

void perform_p_merge_sort(int arr[], int size){

    vector<int> array(size);

    for(int i = 0; i < size; i++){
        array[i] = arr[i];
    }

    double start = omp_get_wtime();

    p_mergesort(array.data(), 0, size - 1);

    double end = omp_get_wtime();

    cout << "Parallel Merge Sort:\n";
    cout << "Time Required: " << end - start << " seconds\n\n";
}

// ===================== MAIN FUNCTION =====================

int main(){

    int SIZE;
    int MAX = 1000;

    cout << "Enter size of array: ";
    cin >> SIZE;

    vector<int> array(SIZE);

    // RANDOM SEED
    srand(time(0));

    // GENERATE RANDOM ARRAY
    for(int i = 0; i < SIZE; i++){
        array[i] = rand() % MAX;
    }

    cout << "\nSorting Started...\n\n";

    sequential_bubble_sort(array.data(), SIZE);

    parallel_bubble_sort(array.data(), SIZE);

    perform_merge_sort(array.data(), SIZE);

    perform_p_merge_sort(array.data(), SIZE);

    return 0;
}
